\section{Implementation Notes}

\textbf{Important caveats for equation-to-code mapping:}

\begin{enumerate}
    \item \textbf{Creator sensitivity $\beta$:} The creator dynamics use $S(\beta x)$ where $\beta$ is a scenario parameter:
    \[
    S(x) = \frac{1}{1+e^{-x}}, \quad \text{applied as } S(\beta(\pi_c - \bar{u}))
    \]
    Default $\beta=15$ makes the IR constraint binding and produces clean regime separation:
    \begin{itemize}
        \item When $\pi_c = 0$: $C^* \approx 0.10$ (collapse possible)
        \item When $\pi_c = \bar{u}$: $C^* = 0.50$ (neutral)
        \item When $\pi_c \gg \bar{u}$: $C^* \to 1$ (full participation)
    \end{itemize}

    The diversion function uses standard logistic ($\beta=1$).

    \item \textbf{Scenario parameterization:} The LaTeX treats $k_A$ as primitive. The JS maps scenario parameter $\alpha$ to $k_A$ via:
    \[
    k_A(\alpha) = 2.0 + 4.0\alpha
    \]
    This is a \textit{scenario family parameterization}, not part of the core model equations.

    \item \textbf{Verification scope:} This document verifies \textit{structural consistency} (equations match code). Outcome claims (collapse, coexistence, etc.) are \textit{hypotheses consistent with parameter choices}; definitive verification requires numerical simulation.
\end{enumerate}

\section{Core Equations}

\subsection{Diversion Propensity}
\[
D(A, \theta) = \sigma(k_A(A - a_0) - k_\theta(\theta - \theta_0))
\]
where $\sigma(x) = 1/(1+e^{-x})$ is standard logistic ($\beta=1$).

Parameters: $k_A = 2.0 + 4.0\alpha$, $k_\theta = 6.0$, $a_0 = 0.30$, $\theta_0 = 0.10$.

\subsection{Creator Dynamics}
\[
\frac{dC}{dt} = (1-C)\rho_C S(\beta(\pi_c - \bar{u})) - C\rho_C S(\beta(\bar{u} - \pi_c))
\]
where $S(x) = 1/(1+e^{-x})$, $\rho_C = 0.15$, $\bar{u} = 0.15$, and $\beta$ is a scenario parameter (default $\beta=15$).

\textbf{Note on equilibrium:} At equilibrium:
\[
(1-C^*)S(\beta(\pi_c - \bar{u})) = C^* S(\beta(\bar{u} - \pi_c))
\]
With $\beta=15$, creators respond sharply to profitability around the $\pi_c = \bar{u}$ threshold.
The sign of $dC/dt$ depends on both $\pi_c$ relative to $\bar{u}$ AND current $C$ relative to $C^*$.

\subsection{Other Equations}
\begin{align*}
\text{Revenue:} \quad & \pi_c = \theta \cdot 0.3 \cdot A + 0.4 \cdot W + m \cdot Q \\
\text{Traffic:} \quad & \frac{dW}{dt} = (1-W)\rho_W(1-D) - W\rho_W D \quad \Rightarrow \quad W^* = 1 - D \\
\text{Quality:} \quad & \frac{dQ}{dt} = (1-Q)[\nu R + \epsilon\max(C-C_{crit},0)] - Q[\lambda_0 + \epsilon\max(C_{crit}-C,0)] \\
\text{AI:} \quad & \frac{dA}{dt} = (1-A)(I + r_A A(1+\gamma Q)) - A\delta_A
\end{align*}

\section{Scenario-by-Scenario Analysis}

\subsection{Scenario 1: scraping-era}
\textbf{Parameters:} $\theta=0.10$, $\alpha=0.35$, $I=0.08$, $m=0.10$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.35) = 3.4$
    \item $D_0 = \sigma(3.4 \times 0.10 - 6.0 \times 0.0) = \sigma(0.34) \approx 0.58$, so $W^* \approx 0.42$
    \item $\pi_{c,0} = 0.10 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.10 \times 0.55 = 0.267 > \bar{u} = 0.15$
    \item Since $D_0 > 0.5$, traffic will decline ($dW < 0$)
    \item As $W \downarrow$, $\pi_c \downarrow$, eventually pushing creators toward exit
\end{itemize}

\textcolor{myorange}{\textbf{Hypothesis:}} Gradual collapse as traffic spiral reduces $\pi_c$ below $\bar{u}$. Quality support is modest. \textit{Requires simulation to confirm timing and trajectory.}

\subsection{Scenario 2: zero-click}
\textbf{Parameters:} $\theta=0.02$, $\alpha=0.90$, $I=0.14$, $m=0.05$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.90) = 5.6$ --- near maximum diversion sensitivity
    \item $D_0 = \sigma(5.6 \times 0.10 - 6.0 \times (-0.08)) = \sigma(1.04) \approx 0.74$, so $W^* \approx 0.26$
    \item $\pi_{c,0} = 0.02 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.05 \times 0.55 = 0.230 > \bar{u}$
    \item Strong $I=0.14$ accelerates $A$ while traffic collapses
\end{itemize}

\textcolor{myorange}{\textbf{Hypothesis:}} Severe, rapid collapse. Near-maximum initial diversion with strong AI growth. \textit{Distinct from scraping-era in severity (higher $\alpha$, higher $k_A$).}

\subsection{Scenario 3: low-quality-trap}
\textbf{Parameters:} $\theta=0.18$, $\alpha=0.25$, $I=0.06$, $m=0.12$

\textbf{Custom ICs:} $C=0.40$, $A=0.40$, $Q=0.40$, $W=0.45$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.25) = 3.0$
    \item $D_0 = \sigma(3.0 \times 0.10 - 6.0 \times 0.08) = \sigma(-0.18) \approx 0.46$, so $W^* \approx 0.54$
    \item $\pi_{c,0} = 0.18 \times 0.3 \times 0.40 + 0.4 \times 0.45 + 0.12 \times 0.40 = 0.250 > \bar{u}$
    \item Starts near critical region, not ``healthy baseline''
\end{itemize}

\textcolor{myorange}{\textbf{Hypothesis:}} Stabilizes at mediocre quality equilibrium. Moderate diversion, only moderate subscription support. System can stabilize with low $Q$ rather than exploding or instantly collapsing. \textit{Requires simulation to confirm stability.}

\subsection{Scenario 4: spotify-news}
\textbf{Parameters:} $\theta=0.40$, $\alpha=0.05$, $I=0.05$, $m=0.35$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.05) = 2.2$
    \item $D_0 = \sigma(2.2 \times 0.10 - 6.0 \times 0.30) = \sigma(-1.58) \approx 0.17$, so $W^* \approx 0.83 > W_0$
    \item $\pi_{c,0} = 0.40 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.35 \times 0.55 = 0.441 \gg \bar{u}$
    \item Since $D_0 < 0.5$, traffic will \textit{increase} ($dW > 0$)
\end{itemize}

\textcolor{mygreen}{\cmark} \textbf{Structurally consistent:} Genuinely low-diversion regime. High $\theta$ ensures strong revenue/reinvestment. Traffic grows toward equilibrium.

\subsection{Scenario 5: ai-winter}
\textbf{Parameters:} $\theta=0.02$, $\alpha=0.70$, $I=0.15$, $m=0.06$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.70) = 4.8$
    \item $D_0 = \sigma(4.8 \times 0.10 - 6.0 \times (-0.08)) = \sigma(0.96) \approx 0.72$, so $W^* \approx 0.28$
    \item $\pi_{c,0} = 0.02 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.06 \times 0.55 = 0.235 > \bar{u}$
    \item Very high $I=0.15$ pushes $A$ hard while traffic collapses
\end{itemize}

\textbf{Mechanism:} High $\alpha$ causes severe traffic collapse. As $Q$ falls, $(1+\gamma Q)$ term weakens.

\textbf{Caveat:} Exogenous $I=0.15$ continues pushing $A$ up even if $Q$ collapses. This produces ``AI boom + ecosystem collapse,'' not guaranteed ``AI winter'' unless $I$ is later dominated by depreciation.

\textcolor{myorange}{\textbf{Hypothesis:}} AI boom followed by ecosystem collapse; AI growth slows but may not fully stagnate. \textit{``Data drought'' effect requires simulation to quantify.}

\subsection{Scenario 6: substack-lifeboat}
\textbf{Parameters:} $\theta=0.06$, $\alpha=0.70$, $I=0.05$, $m=0.50$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.70) = 4.8$
    \item $D_0 \approx 0.74$, so $W^* \approx 0.26$ (traffic loss is high)
    \item $\pi_{c,0} = 0.06 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.50 \times 0.55 = 0.482 \gg \bar{u}$
\end{itemize}

\textbf{Key insight:} High $m=0.50$ means subscription term dominates revenue:
\[
\pi_c \approx 0.50 \times Q
\]
Even at $W=0.20$, $Q=0.40$: subscription term $= 0.20 > \bar{u}$.

\textcolor{mygreen}{\cmark} \textbf{Structurally consistent:} Diversion is high (traffic loss), but $m$ is so large that creators remain viable via $mQ$. The ``lifeboat'' effect: subscriptions insulate from traffic collapse.

\subsection{Scenario 7: journalism-graveyard}
\textbf{Parameters:} $\theta=0.08$, $\alpha=0.60$, $I=0.08$, $m=0.08$, $\beta=15$ (default)

\textbf{Custom ICs:} $C=0.20$, $A=0.55$, $Q=0.15$, $W=0.10$

\textbf{Critical calculation:}
\begin{itemize}
    \item $k_A = 2 + 4(0.60) = 4.4$
    \item $D_0 = \sigma(4.4 \times 0.25 - 6.0 \times (-0.02)) = \sigma(1.22) \approx 0.77$, so $W^* \approx 0.23$
    \item $\pi_{c,0} = 0.08 \times 0.3 \times 0.55 + 0.4 \times 0.10 + 0.08 \times 0.15 = 0.065 \ll \bar{u} = 0.15$
\end{itemize}

\textbf{Key insight:} $\pi_{c,0} = 0.065 \ll \bar{u}$ decisively violates IR. With $\beta=15$:
\[
S(15 \times (0.065 - 0.15)) = S(-1.275) \approx 0.22
\]
This means entry rate is very low while exit rate is high. Combined with $C_0 = 0.20 < C_{crit} = 0.35$, quality decays, further reducing $\pi_c$.

\textcolor{myorange}{\textbf{Hypothesis:}} Decisive collapse. The ``can't climb back'' dynamics are now structurally plausible with deep sub-threshold ICs.

\textit{Designed to demonstrate true hysteresis-style collapse.}

\subsection{Scenario 8: death-by-1000-cuts}
\textbf{Parameters:} $\theta=0.08$, $\alpha=0.20$, $I=0.04$, $m=0.12$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.20) = 2.8$
    \item $D_0 = \sigma(2.8 \times 0.10 - 6.0 \times (-0.02)) = \sigma(0.40) \approx 0.60$, so $W^* \approx 0.40$
    \item $\pi_{c,0} = 0.08 \times 0.3 \times 0.40 + 0.4 \times 0.50 + 0.12 \times 0.55 = 0.076 > \bar{u}$
\end{itemize}

\textbf{Key insight:} $D_0 \approx 0.60$ is only slightly above 0.5, producing slow decline. Very low $I=0.04$ means AI grows extremely slowly, so $D$ increases very slowly. The ``boiled frog'' scenario where decline is so gradual it's imperceptible.

\textcolor{myorange}{\textbf{Hypothesis:}} Very slow erosion that is hard to notice until too late. \textit{Distinct from other collapse scenarios in speed due to low $I$.}

\subsection{Scenario 9: marshall-plan}
\textbf{Parameters:} $\theta=0.42$, $\alpha=0.05$, $I=0.04$, $m=0.30$

\textbf{Damaged ICs:} $C=0.30$, $A=0.45$, $Q=0.30$, $W=0.25$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.05) = 2.2$
    \item $D_0 = \sigma(2.2 \times 0.15 - 6.0 \times 0.32) = \sigma(-1.59) \approx 0.17$, so $W^* \approx 0.83$
    \item $\pi_{c,0} = 0.42 \times 0.3 \times 0.45 + 0.4 \times 0.25 + 0.30 \times 0.30 = 0.247 > \bar{u}$
\end{itemize}

Since $D_0 < 0.5$, traffic will recover ($dW > 0$) from damaged $W_0=0.25$ toward $W^* \approx 0.83$.

\textcolor{mygreen}{\cmark} \textbf{Structurally consistent:} Very high $\theta$ and very low $\alpha$ produce low diversion. Starts in genuinely damaged state but with parameters that move $W$ upward and revenue toward viability. This is what makes it a ``policy recovery'' scenario.

\subsection{Scenario 10: boom-bust}
\textbf{Parameters:} $\theta=0.22$, $\alpha=0.15$, $I=0.10$, $m=0.18$

\textbf{Custom ICs:} $C=0.35$, $A=0.45$, $Q=0.45$, $W=0.45$

\textbf{Initial state analysis:}
\begin{itemize}
    \item $k_A = 2 + 4(0.15) = 2.6$
    \item $D_0 = \sigma(2.6 \times 0.15 - 6.0 \times 0.12) = \sigma(-0.33) \approx 0.42$, so $W^* \approx 0.58$
    \item $\pi_{c,0} = 0.22 \times 0.3 \times 0.45 + 0.4 \times 0.45 + 0.18 \times 0.45 = 0.291 > \bar{u}$
    \item $W_0 = 0.45 \approx W^*$ (near traffic equilibrium)
    \item $C_0 = 0.35 \approx C_{crit}$ (near creator threshold)
\end{itemize}

\textbf{Key insight:} Tuned close to the knife-edge. $W_0 \approx W_0^*$ and $C_0 \approx C_{crit}$, which is where delayed adjustments \textit{can} create overshoot patterns.

\textbf{Important:} If you truly want reliable oscillations, add one calibration lever (speed mismatch: raise $\rho_W$, lower $\rho_C$)---otherwise this remains a plausible-but-empirical regime.

\textcolor{myorange}{\textbf{Hypothesis:}} Near-threshold feedback with potential oscillation dynamics. \textit{Oscillations not guaranteed without speed mismatch.}
